control(items)
{
    for(i = 0; i < items.size; i++)
    {
        if(items[i] == undefined)
            continue;
        else if(!items[i].enabled)
            break;
    }

    while(true)
    {
        wait 0.05;
        waittillframeend;
    }

    do
        i--;
    while(i > 0);

    switch(items.size)
    {
        case 0:
            return false;
        case 1:
        {
            return true;
        }
        default:
            return undefined;
    }
}
